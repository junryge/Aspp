# 3) 50초 집계
# ============================================================
def aggregate(files):
    rep = defaultdict(set)
    jam = defaultdict(set)
    ht = defaultdict(set)
    seen_by_file = []
    for p in files:
        seen = set()
        with open(p, encoding="utf-8-sig", newline="") as f:
            for r in csv.DictReader(f):
                v = (r.get("VEHICLE") or "").strip()
                if not v or (VEHICLE_PREFIX and not v.startswith(VEHICLE_PREFIX)):
                    continue
                t = parse_time(r.get("_time"))
                if not t:
                    continue
                b = bucket_of(t)
                st = (r.get("STATUS") or "").strip()
                rep[b].add(v)
                seen.add(v)
                if st == "7":
                    jam[b].add(v)
                elif st == "8":
                    ht[b].add(v)
        seen_by_file.append((p, seen))
    fleet_auto = max((len(s) for _, s in seen_by_file), default=0)
    fleet = FLEET_SIZE or fleet_auto
    table = {}
    for b in sorted(rep):
        table[b] = {"rep": len(rep[b]), "miss": max(0, fleet - len(rep[b])),
                    "jam": len(jam[b]), "ht": len(ht[b])}
    return table, fleet

# 4) 판정
# ============================================================
def level(mm, jj, hh):
    if mm >= TH_CONFIRM_MISS or hh >= TH_CONFIRM_HT:
        return "확정"
    if mm >= TH_ALERT_MISS or jj >= TH_ALERT_JAM:
        return "경보"
    if mm >= TH_PRE_MISS or jj >= TH_PRE_JAM:
        return "전조"
    return "정상"


def judge(alarms, table):
    out = []
    for a in alarms:
        t = a["time"]
        s = t - timedelta(minutes=LOOKBACK_MIN)
        w = [v for b, v in table.items() if s <= b <= t]
        if not w:
            out.append({**a, "n": 0, "miss": "", "jam": "", "ht": "", "oht": "데이터없음", "판정": "판정불가"})
            continue
        mm = max(v["miss"] for v in w)
        jj = max(v["jam"] for v in w)
        hh = max(v["ht"] for v in w)
        lv = level(mm, jj, hh)
        verdict = {"확정": "진짜(확정)", "경보": "진짜(확정)", "전조": "진짜(전조)", "정상": "관찰"}[lv]
        out.append({**a, "n": len(w), "miss": mm, "jam": jj, "ht": hh, "oht": lv, "판정": verdict})
    return out


def load_truth(path):
    rows = read_text_table(path, "시간")
    ts = [parse_time(r.get("시간")) for r in rows]
    return [t for t in ts if t]


def score_truth(res, truth, before_min=60, after_min=10):
    """알람 시각이 [사건-60분, 사건+10분] 안이면 그 사건을 잡은 것으로 본다."""
    for r in res:
        r["정답"] = "진짜사건" if any(tt - timedelta(minutes=before_min) <= r["time"] <= tt + timedelta(minutes=after_min)
                                  for tt in truth) else "사건없음"
    return res

# main
# ============================================================
def main():
    ap = argparse.ArgumentParser(description="스코어 알람 → 과거 10분 OHT 판정")
    ap.add_argument("--score", required=True, help="실시간 스코어 CSV (시간, 등급, 종합점수, 알람 …)")
    ap.add_argument("--truth", help="실제 사건 시각 CSV (헤더 '시간') — 있으면 정밀도 계산")
    ap.add_argument("--out", default=os.path.join(HERE, "out"))
    ap.add_argument("--raw", default=os.path.join(HERE, "oht_raw"))
    ap.add_argument("--test", action="store_true", help="연결 확인 (5줄만)")
    ap.add_argument("--dry-run", action="store_true", help="조회 구간·쿼리만 출력")
    args = ap.parse_args()

    alarms = load_alarms(args.score)
    print(f"[1] 알람 발생 {len(alarms)}건  " + " · ".join(
        f"{g} {n}" for g, n in Counter(a['grade'] for a in alarms).items()))
    if not alarms:
        sys.exit("알람이 없습니다 — 스코어 파일의 '시간' '등급' 컬럼을 확인하세요")
    chunks = windows_for(alarms)
    total_min = sum((e - s).total_seconds() for s, e in chunks) / 60
    print(f"[2] 조회 구간 {len(chunks)}개 · 합계 {total_min/60:.1f}시간 "
          f"({alarms[0]['time']:%m-%d %H:%M} ~ {alarms[-1]['time']:%m-%d %H:%M})")

    if args.dry_run:
        for s, e in chunks:
            print("   ", build_query(s, e))
        return

    key = api_key()
    if args.test:
        s, e = chunks[0]
        q = build_query(s, s + timedelta(minutes=1)) + " | limit 5"
        p = os.path.join(HERE, "_test.csv")
        print("   쿼리:", q)
        fetch(q, p, key)
        with open(p, encoding="utf-8-sig") as f:
            print(f.read()[:1500])
        print("   ↑ VEHICLE, _time, STATUS 가 보이면 정상입니다")
        return

    os.makedirs(args.raw, exist_ok=True)
    files = []
    for i, (s, e) in enumerate(chunks, 1):
        p = os.path.join(args.raw, f"oht_{s:%Y%m%d_%H%M%S}_{e:%H%M%S}.csv")
        files.append(p)
        if os.path.exists(p) and os.path.getsize(p) > 0:
            continue
        t0 = time.time()
        fetch(build_query(s, e), p, key)
        print(f"   {i}/{len(chunks)}  {s:%m-%d %H:%M}~{e:%H:%M}  "
              f"{os.path.getsize(p)/1e6:.1f}MB  {time.time()-t0:.0f}초")

    print("[3] 50초 집계")
    table, fleet = aggregate(files)
    print(f"    구간 {len(table)}개 · 전체 차량 {fleet}대")
    os.makedirs(args.out, exist_ok=True)
    with open(os.path.join(args.out, "oht_50s.csv"), "w", encoding="utf-8-sig", newline="") as f:
        w = csv.writer(f)
        w.writerow(["시각", "보고차량", "미보고", "JAM", "HT_STOP"])
        for b, v in sorted(table.items()):
            w.writerow([b.strftime("%Y-%m-%d %H:%M:%S"), v["rep"], v["miss"], v["jam"], v["ht"]])

    print("[4] 알람 판정")
    res = judge(alarms, table)
    if args.truth:
        res = score_truth(res, load_truth(args.truth))
    cols = ["알람시각", "등급", "점수", "알람컬럼", "과거10분_구간수", "미보고_최대", "JAM_최대",
            "HT_STOP_최대", "OHT단계", "판정"] + (["정답"] if args.truth else [])
    with open(os.path.join(args.out, "알람판정.csv"), "w", encoding="utf-8-sig", newline="") as f:
        w = csv.writer(f)
        w.writerow(cols)
        for r in res:
            row = [r["time"].strftime("%Y-%m-%d %H:%M"), r["grade"], r["score"], r["alarm_col"],
                   r["n"], r["miss"], r["jam"], r["ht"], r["oht"], r["판정"]]
            if args.truth:
                row.append(r["정답"])
            w.writerow(row)

    lines = [f"알람 {len(res)}건 · 기준: 과거 {LOOKBACK_MIN}분 · {BUCKET_SEC}초 집계 · 전체 차량 {fleet}대", ""]
    lines.append("등급 × 판정")
    for g in ("경계", "위험", "초위험"):
        c = Counter(r["판정"] for r in res if r["grade"] == g)
        if c:
            lines.append(f"  {g:4} " + " · ".join(f"{k} {v}" for k, v in c.most_common()))
    if args.truth:
        lines += ["", "정답 비교 (알람이 사건 60분 전 ~ 10분 후 안이면 진짜사건)"]
        for name, sel in (("기존 알람 전체", lambda r: True),
                          ("OHT 진짜 판정만", lambda r: r["판정"].startswith("진짜"))):
            rs = [r for r in res if sel(r)]
            tp = sum(r["정답"] == "진짜사건" for r in rs)
            prec = tp / len(rs) * 100 if rs else 0
            lines.append(f"  {name:12} 알람 {len(rs):4}건 · 진짜사건 {tp:4}건 · 정밀도 {prec:5.1f}%")
        miss_true = [r for r in res if r["정답"] == "진짜사건" and r["판정"] == "관찰"]
        lines.append(f"  진짜사건인데 '관찰'로 판정한 알람 {len(miss_true)}건 (놓친 것 — 확인 필요)")
    txt = "\n".join(lines)
    print(txt)
    with open(os.path.join(args.out, "요약.txt"), "w", encoding="utf-8") as f:
        f.write(txt + "\n")
    print(f"\n완료 → {args.out}")


if __name__ == "__main__":
    main()
